      <script>
        // base/plain (styles spec "Caption presets"): the active word is bold, emphasis words are
        // 1.2x larger, read and unread words look the same. Lines come from the engine (never
        // re-wrapped). The box is placed without measuring the DOM: the caption font is monospaced,
        // so a line of maxCharsPerLine characters has a known width, and a chunk's height is its line
        // count times the line height. All state changes are timeline sets at word times.
        (function () {
          var mount = vox.mount('captions');
          var root = mount.root;
          var v = mount.vars;
          var W = 1080;
          var H = 1920;
          var side = (v.safeArea.rightPct / 100) * W;
          var usable = W - 2 * side;
          // IBM Plex Mono advance is 0.6 em; leave room for 1.2x emphasis words
          var size = Math.min(W * 0.065, usable / (Math.max(1, v.maxCharsPerLine) * 0.6 * 1.2));
          var lineH = size * 1.2 * 1.25;
          var limit = (1 - v.safeArea.bottomPct / 100) * H;
          var centre = (v.position.yPct / 100) * H;
          var normal =
            getComputedStyle(root).getPropertyValue('--tok-font-caption-weight').trim() || '400';
          var tl = gsap.timeline({ paused: true });
          for (var i = 0; i < v.chunks.length; i++) {
            var c = v.chunks[i];
            var el = document.createElement('div');
            el.className = 'c-chunk';
            var h = c.lines.length * lineH;
            var top = Math.max(0, Math.min(centre - h / 2, limit - h));
            el.style.left = side + 'px';
            el.style.width = usable + 'px';
            el.style.top = top + 'px';
            el.style.fontSize = size + 'px';
            el.style.lineHeight = lineH + 'px';
            for (var l = 0; l < c.lines.length; l++) {

              line.className = 'c-line';
